const compressImage = (file) => {
  return new Promise((resolve) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = (event) => {
      const img = new Image();
      img.src = event.target.result;
      img.onload = () => {
        const canvas = document.createElement('canvas');
        const MAX_WIDTH = 800; 
        const scaleSize = MAX_WIDTH / img.width;
        canvas.width = MAX_WIDTH;
        canvas.height = img.height * scaleSize;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
        resolve(canvas.toDataURL('image/jpeg', 0.6));
      };
    };
  });
};

const Dashboard = ({ entries, companyStats }) => {
  const monthlyData = useMemo(() => {
    const driverStats = DRIVERS.map(d => {
      const dEntries = entries.filter(e => e.driverId === d.id);
      const transportRevenue = dEntries.reduce((sum, e) => sum + (Number(e.transportPrice) || 0), 0);
      const income = dEntries.reduce((sum, e) => sum + (e.tripPay || 0), 0);
      const expense = dEntries.reduce((sum, e) => sum + (e.fuelCost || 0) + (e.otherCost || 0), 0);
      const fuelExpense = dEntries.reduce((sum, e) => sum + (e.fuelCost || 0), 0);
      const distance = dEntries.reduce((sum, e) => sum + (e.distance || 0), 0);
      const fuel = dEntries.reduce((sum, e) => sum + (e.fuelLitres || 0), 0);
      const consumption = fuel > 0 ? (distance / fuel).toFixed(2) : 0;
      return { name: d.nickname, transportRevenue, income, expense, fuelExpense, consumption };
    });
    return driverStats;
  }, [entries]);

  const maxVal = Math.max(...monthlyData.map(d => Math.max(d.income, d.expense)), 1000);

  return (
    <div className="space-y-6 animate-fade-in-up">
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <Card className="p-6">
          <h3 className="text-lg font-bold text-gray-800 mb-4 flex items-center">
            <BarChart3 className="w-5 h-5 mr-2 text-blue-600" /> เปรียบเทียบรายรับ-รายจ่าย (เดือนนี้)
          </h3>
          <div className="space-y-4">
            {monthlyData.map(d => (
              <div key={d.name} className="space-y-1">
                <div className="flex justify-between text-sm font-medium">
                  <span>{d.name}</span>
                </div>
                <div className="h-4 w-full bg-gray-100 rounded-full overflow-hidden flex">
                  <div style={{ width: `${(d.income / maxVal) * 100}%` }} className="h-full bg-green-500" title={`รายรับ(ค่าเที่ยว): ${d.income.toLocaleString()}`}></div>
                </div>
                <div className="h-4 w-full bg-gray-100 rounded-full overflow-hidden flex">
                  <div style={{ width: `${(d.expense / maxVal) * 100}%` }} className="h-full bg-red-400" title={`รายจ่ายรวม: ${d.expense.toLocaleString()}`}></div>
                </div>
                <div className="flex justify-between text-xs text-gray-500">
                  <span>รับ: {d.income.toLocaleString()}</span>
                  <span>จ่ายรวม: {d.expense.toLocaleString()} (น้ำมัน: {d.fuelExpense.toLocaleString()})</span>
                </div>
                <div className="text-[11px] text-purple-600 font-medium text-right mt-0.5">
                  ยอดค่าขนส่งที่ทำได้: {d.transportRevenue.toLocaleString()} บาท
                </div>
              </div>
            ))}
          </div>
        </Card>

        <Card className="p-6">
          <h3 className="text-lg font-bold text-gray-800 mb-4 flex items-center">
            <Gauge className="w-5 h-5 mr-2 text-orange-500" /> อัตราสิ้นเปลือง (กม./ลิตร)

          <div className="flex items-end justify-around h-40 space-x-4">
